find_package(acados REQUIRED)

# The acados solver generated by codegen/generate_solver.py. It is C, so it is built as its own library without the C++
# options every module gets (-std=c++20 is an error for C under -Werror) or the warnings (it is generated code).
file(GLOB_RECURSE solver_sources "${CMAKE_CURRENT_SOURCE_DIR}/solver/*.c")
add_library(ModelledMPCWalkPathSolver STATIC ${solver_sources})
get_target_property(solver_options ModelledMPCWalkPathSolver COMPILE_OPTIONS)
list(
  REMOVE_ITEM
  solver_options
  -std=c++20
  -Wall
  -Wpedantic
  -Wextra
  -Werror
)
set_target_properties(
  ModelledMPCWalkPathSolver PROPERTIES COMPILE_OPTIONS "${solver_options};-w" POSITION_INDEPENDENT_CODE ON
)
target_include_directories(ModelledMPCWalkPathSolver SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/solver")
target_link_libraries(ModelledMPCWalkPathSolver PUBLIC acados::acados)

# Build our NUClear module
nuclear_module(LIBRARIES ModelledMPCWalkPathSolver)
