42  /// Runs the mujoco_playground K1 joystick walk policy (79-obs / 22-action ONNX, see
43  /// NUSim docs/OBS_ACTION_CONTRACT.md) on the robot side and streams the resulting
44  /// joint targets to the platform as BoosterLowCmd (rt/joint_ctrl, CUSTOM mode). This
45  /// replaces skill::Walk's Move() RPC path: locomotion inference lives here, and the
46  /// robot/simulator only has to track servo joint commands.
47  ///
48  /// The observation carries no base linear velocity. It used to (82 obs), sourced from a
49  /// finite-difference of rt/odometer_state, but there is no measured linear velocity on
50  /// the real K1 in CUSTOM mode -- so the policy is now trained with linvel as a
51  /// critic-only privileged quantity and the deployment side has nothing to estimate.
52  class SimK1Walk : public ::extension::behaviour::BehaviourReactor {
53  public:
54  static constexpr std::size_t JOINT_COUNT = 22; // SDK JointIndexK1 serial order
55  static constexpr std::size_t OBS_DIM = 79;
56 
57  explicit SimK1Walk(std::unique_ptr<NUClear::Environment> environment);
58 
59  private:
60  struct Config {
61  std::string model_path;
62  /// Inference device: "cpu", or "gpu" for ONNX Runtime's TensorRT EP (fp16). gpu falls
63  /// back to cpu when it cannot load (no CUDA device, driver mismatch, ...).
64  std::string device = "cpu";
65  /// @brief Hz, gait phase advance rate (training randomizes U(1.25, 1.75))
66  double gait_frequency = 1.5;
67  /// @brief command norm below which the phase observation pins to [pi, pi]
68  double stand_threshold = 0.01;
69  /// @brief PD gains for the two head joints (the policy does not own the head)
70  double head_kp = 10.0;
71  double head_kd = 0.5;
72  /// @brief Velocity command applied while performing an in-walk kick
73  Eigen::Vector3d kick_velocity = Eigen::Vector3d::Zero();
74  /// @brief How long to drive the kick velocity before reporting the kick as done
75  NUClear::clock::duration kick_duration{};
76  /// @brief Training-time actuation, JointIndexK1 order (see SimK1Walk.yaml)
77  std::array<double, JOINT_COUNT> kp{};
78  std::array<double, JOINT_COUNT> kd{};
79  std::array<double, JOINT_COUNT> action_scale_joint{};
80  std::array<double, JOINT_COUNT> default_pose{};
81  /// @brief Hard joint ranges of the trained MuJoCo model, JointIndexK1 order. The
82  /// commanded position is clamped to these: MuJoCo's joint constraint silently
83  /// absorbs an out-of-range target, but on the robot it is a leg torquing into a
84  /// mechanical stop.
85  std::array<double, JOINT_COUNT> joint_lower{};
86  std::array<double, JOINT_COUNT> joint_upper{};
87  } cfg;
88 
89  /// ONNX Runtime inference backend
90  std::unique_ptr<utility::onnx::ONNXRuntime> policy{};
91  bool model_loaded = false;
92 
93  /// Previous raw network output (part of the observation)
94  std::array<float, JOINT_COUNT> last_action{};
95  /// Per-foot gait phase, initialized anti-phase [0, pi]
96  std::array<double, 2> phase{0.0, 3.141592653589793};
97 
98  /// Wall-clock of the previous policy tick, so the gait phase advances on the period
99  /// that actually elapsed rather than on a hardcoded 0.02 s. Every<50, Per<seconds>>
100  /// is best-effort on an Orin also running YOLO: at a true 40 Hz a nominal 1.5 Hz
101  /// gait actually runs at 1.20 Hz, below the U(1.25, 1.75) training range.
102  bool have_last_tick = false;
103  NUClear::clock::time_point last_tick_time{};
104 
105  /// Monotonic tick counter and the robot's last reported motion mode, both logged with
106  /// every observation: statistics over a log that mixes CUSTOM-mode walking with
107  /// frozen non-CUSTOM ticks describe a policy shouting at a robot that isn't listening.
108  std::uint64_t tick = 0;
109  int last_mode = -1;
110 
111  /// Latest clamped head target (yaw, pitch) from BoosterHeadRot
112  Eigen::Vector2d head_target = Eigen::Vector2d::Zero();
113 
114  /// Time the current in-walk kick started
115  NUClear::clock::time_point kick_start_time{};
116 
117  /// Last emitted walk state, to avoid re-emitting unchanged states at 50 Hz
118  int last_walk_state = -1;
119 
120  void reset_policy_state();
121  };
