49  /// @brief Stores the offset, attraction, and minimum x for a robot's formation slot
50  struct RobotSlot {

65  /// @brief Stores configuration values
66  struct Config {
67  /// @brief Formation slots per game mode, keyed by mode name then player ID
68  std::map<std::string, std::map<int, RobotSlot>> modes;
69  /// @brief Translational/angular error threshold (metres/radians) within which Support
70  /// considers itself arrived at its target position. Passed to WalkToFieldPosition as
71  /// a per-task override (so only Support gets this radius) and reused locally to gate
72  /// when Support starts looking at the ball.
73  double stop_threshold = 0.0;
74  /// @brief Wider hysteresis threshold once stopped, passed as WalkToFieldPosition's
75  /// stopped-threshold override.
76  double stopped_threshold = 0.0;
77  } cfg;
78 
79  /// @brief Resolve this robot's formation slot for the given mode, falling back to the
80  /// normal_play formation when the mode is not defined. Returns nullptr when neither
81  /// the mode nor the fallback defines a slot for this robot (e.g. Formation.yaml has
82  /// not been loaded yet), so callers never dereference a past-the-end iterator.
83  /// @param mode_name the formation mode to look up
84  /// @param player_id the robot's player ID
85  /// @return pointer to the matching slot, or nullptr if none applies
86  const RobotSlot* find_slot(const std::string& mode_name, int player_id) const;
87 
88  /// @brief Compute what this robot's support position would be right now, given the current ball,
89  /// field, game state and formation config - regardless of whether this robot is actually
90  /// assigned the Support purpose. Used both by the active Support behaviour (to know where to
91  /// walk) and by an always-on reactor that lets NUsight show a live preview of this position
92  /// even when some other robot is currently the one supporting.
93  /// @return the computed field-space position (x, y, 0), or std::nullopt if no formation slot applies
94  /// (e.g. Formation.yaml has not been loaded yet)
95  std::optional<Eigen::Vector3d> calculate_support_position(
96  const std::shared_ptr<const message::localisation::Ball>& ball,
97  const message::localisation::Field& field,
98  const message::input::GameState& game_state,
99  const message::support::GlobalConfig& global_config,
100  const message::support::FieldDescription& fd) const;
101 
102  public:
103  /// @brief Called by the powerplant to build and setup the Support reactor.
104  explicit Support(std::unique_ptr<NUClear::Environment> environment);
105  };
